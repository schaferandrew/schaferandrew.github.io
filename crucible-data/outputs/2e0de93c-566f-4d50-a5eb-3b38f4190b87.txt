# Question 1: Compound Interest Calculation
def calculate_compound_interest(principal, annual_rate, times_compounded_per_year, years):
    rate_per_period = annual_rate / (times_compounded_per_year * 100)
    total_periods = times_compounded_per_year * years
    amount = principal * (1 + rate_per_period) ** total_periods
    return round(amount, 2)

# Question 2: Flooring Calculation
def calculate_flooring_boxes(total_sq_ft, box_size_sq_ft, extra_percentage):
    total_needed_sq_ft = total_sq_ft * (1 + extra_percentage / 100)
    boxes_needed = -(-total_needed_sq_ft // box_size_sq_ft)  # Ceiling division
    return int(boxes_needed)

# Question 3: Gas Cost Calculation
def calculate_gas_cost(total_miles, miles_per_gallon, gas_price_per_gallon):
    gallons_needed = total_miles / miles_per_gallon
    total_cost = gallons_needed * gas_price_per_gallon
    return round(total_cost, 2)

# Question 4: Dinner Splitting with Tip Calculation
def calculate_dinner_split(pre_tax_amount, tip_percentage, number_of_people):
    tip_amount = pre_tax_amount * (tip_percentage / 100)
    total_bill = pre_tax_amount + tip_amount
    amount_per_person = total_bill / number_of_people
    return round(amount_per_person, 2)

# Answers to the questions
answers = [
