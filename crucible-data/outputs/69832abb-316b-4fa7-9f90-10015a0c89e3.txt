### Key Requirements Summary
1. **Article Status & Review Flow**:
   - Articles start as `draft`. Submitting a draft moves it to `in_review`; approval publishes it.
   - Unapproved drafts must remain private to editors only.

2. **API Endpoints**:
   - `POST /api/articles`: Accepts optional `"status": "draft"` to move to review.
   - `POST /api/articles/:slug/submit`: Submit draft to `in_review`.
   - `POST /api/articles/:slug/approve`: Approve draft to `published`.
   - `POST /api/articles/:slug/reject`: Reject draft with feedback.

3. **Review Queue**:
   - `GET /api/articles/review-queue`: Returns editors' pending reviews.

4. **User Articles**:
   - `GET /api/user/articles`: Returns user's articles by status.

5. **Editor Configuration**:
   - Use environment variable `EDITORS` to set allowed editors (e.g., `"editor1,editor2"`).

6. **Compatibility**:
   - Ensure existing clients (especially the mobile client) continue to function.
   - Keep unapproved drafts private.

app.post('/api/articles', async (req, res) => {
  const { status, ...articleData } = req.body;
  const userId = req.user.id;
  const userEditor = await getUserEditor(userId); // fetch from DB
  if (!userEditor.editors.includes(auth.editor)) {
    return res.status(403).json({ error: 'Forbidden' });
  }
  if (status && ['draft', 'in_review', 'published'].includes(status)) {
    const createdAt = new Date();
    const createdArticle = await Article.create({
      ...articleData,
      status,
      reviewedBy: userId, // optional
      createdAt
    });
    if (status === 'draft') {
      return res.status(201).json({ article: createdArticle });
    }
    if (status === 'in_review') {
      return res.status(202).json({ article: createdArticle });
    }
    if (status === 'published') {
      return res.status(200).json({ article: createdArticle });
    }
  }
  res.status(400).json({ error: 'Invalid request' });
});

// POST /api/articles/:slug/submit
app.post('/api/articles/:slug/submit', checkAuthAndEditor(['draft']), async (req, res) => {
  // move draft to in_review
  const article = await Article.findOne({ slug: req.params.slug, status: 'draft' });
  if (!article) {
    return res.status(404).json({ error: 'Not found' });
  }
  article.status = 'in_review';
  await article.save();
  return res.status(202).json({ article });
});

// POST /api/articles/:slug/approve
app.post('/api/articles/:slug/approve', checkAuthAndEditor(['in_review']), async (req, res) => {
  const article = await Article.findOne({ slug: req.params.slug, status: 'in_review' });
  if (!article) {
    return res.status(404).json({ error: 'Not found' });
  }
  article.status = 'published';
  await article.save();
  return res.json({ article });
});

// POST /api/articles/:slug/reject
app.post('/api/articles/:slug/reject', checkAuthAndEditor(['in_review']), async (req, res) => {
  const { review } = req.body;
  const article = await Article.findOne({ slug: req.params.slug, status: 'in_review' });
  if (!article) {
    return res.status(404).json({ error: 'Not found' });
  }
  article.status = 'draft';
  article.reviewComment = review.comment;
  await article.save();
  return res.json({ article });
});

// GET /api/articles/review-queue
app.get('/api/articles/review-queue', checkAuthAndEditor(['in_review']), async (req, res) => {
  const articles = await Article.find({ status: 'in_review' }).sort('createdAt ASC');
  return res.json({ articles, articlesCount: articles.length });
});

// GET /api/user/articles
app.get('/api/user/articles', checkAuthAndEditor(['draft', 'in_review', 'published']), async (req, res) => {
  const { status } = req.query;
  const filter = status ? { status } : {};
  const articles = await Article.find(filter).sort('createdAt DESC');
  return res.json(articles);
});
