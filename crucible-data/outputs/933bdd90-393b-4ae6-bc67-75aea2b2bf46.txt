# Puzzle A: 9x9 Sudoku

def is_valid(board, row, col, num):
    # Check if the number is not repeated in the current row
    for x in range(9):
        if board[row][x] == num:
            return False

    # Check if the number is not repeated in the current column
    for x in range(9):
        if board[x][col] == num:
            return False

    # Check if the number is not repeated in the 3x3 box
    startRow = row - row % 3
    startCol = col - col % 3
    for i in range(3):
        for j in range(3):
            if board[i + startRow][j + startCol] == num:
                return False

    return True

def solve_sudoku(board):
    empty = find_empty_location(board)
    if not empty:
        return True  # Puzzle solved
    row, col = empty

    for num in range(1, 10):
        if is_valid(board, row, col, str(num)):
            board[row][col] = str(num)

            if solve_sudoku(board):
                return True

            # Undo the current cell for backtracking
            board[row][col] = '.'

    return False

def find_empty_location(board):
    for i in range(9):
        for j in range(9):
            if board[i][j] == '.':
                return (i, j)
    return None

                return False

    knight_moves = [(2, 1), (1, 2), (-1, 2), (-2, 1),

        if 0 <= new_row < 9 and 0 <= new_col < 9 and board[new_row][new_col] == num:

# Puzzle C: 6x6 Irregular Sudoku

def is_valid_irregular(board, row, col, num):
    # Check if the number is not repeated in the current row
    for x in range(6):
        if board[row][x] == num:
            return False

    # Check if the number is not repeated in the current column
    for x in range(6):
        if board[x][col] == num:
            return False

    # Check if the number is not repeated in the region
    region_map = [
        ['U', 'U', 'U', 'V', 'V', 'V'],
        ['W', 'U', 'U', 'V', 'V', 'X'],
        ['W', 'W', 'U', 'V', 'X', 'X'],
        ['Y', 'W', 'X', 'X', 'X', 'Z'],
        ['Y', 'W', 'W', 'Z', 'Z', 'Z'],
        ['Y', 'Y', 'Y', 'Z', 'Z', '.']
    ]
    region = region_map[row][col]
    for i in range(6):
        for j in range(6):
            if region_map[i][j] == region and board[i][j] == num:
                return False

    return True

def solve_irregular_sudoku(board, region_map):
    empty = find_empty_location(board)
    if not empty:
        return True  # Puzzle solved
    row, col = empty

    for num in range(1, 7):
        if is_valid_irregular(board, row, col, str(num)):
            board[row][col] = str(num)

            if solve_irregular_sudoku(board, region_map):
                return True

            # Undo the current cell for backtracking
            board[row][col] = '.'

    return False
