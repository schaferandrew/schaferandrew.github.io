  const [success, setSuccess] = useState(false);

      setMessage("All fields are required.");
      return;

    // Client-side email/name normalization
    const normalizedEmail = email.normalizeEmail();
    const normalizedMessage = message.trim().replace(/\s+/g, " ");

    try {
      const res = await fetch("/api/leads", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ name, email: normalizedEmail, company, message: normalizedMessage }),
      });
      const data = await res.json();
      if (res.ok) {
        setSuccess(true);
        // Optionally reset form
        setName(""); setEmail(""); setCompany(""); setMessage("");
      } else {
        // Server returned errors
        if (res.headers.get("content-type")?.includes("application/json")) {
          // Extract errors from JSON
          const err = await res.json();
          Object.keys(err.errors || {}).forEach(key => {
            // e.g., "name.error", "email.error" ...
            const field = key.replace(/^([a-z]+)-error$/, "$1");
            // You can set field-specific errors here if you want
          });

        placeholder="Your name"
        aria-invalid={!!errors.name}
        aria-describedby={errors.name ? "name-error" : undefined}
      />
      <input
        value={email}
        onChange={e => setEmail(e.target.value)}
        placeholder="Email"
        aria-invalid={!!errors.email}

      />

        value={company || ""}

        aria-describedby={errors.company ? "company-error" : undefined}
